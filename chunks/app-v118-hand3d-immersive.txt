// V10.18 — HAND 3D MODO IMERSIVO / FOCO MOBILE
let v118Immersive=false;
let v118ControlsVisible=true;

function v118SyncModeButtons(){
  document.querySelectorAll('#v118ImmersiveBar [data-handmode]').forEach(btn=>{
    btn.classList.toggle('active',btn.dataset.handmode===v116HandMode);
  });
}

function v118SyncControlsVisibility(){
  document.body.classList.toggle('v118-hand-controls-hidden',!v118ControlsVisible);
  const toggle=document.getElementById('v118ToggleControls');
  if(toggle) toggle.textContent=v118ControlsVisible?'Ocultar controles':'Mostrar controles';
}

function v118EnterImmersive(){
  if(v118Immersive)return;
  v118Immersive=true;
  v118ControlsVisible=true;
  document.body.classList.add('v118-hand-immersive');
  v118SyncModeButtons();
  v118SyncControlsVisibility();
  try{renderer?.setSize?.(stage.clientWidth,stage.clientHeight,false);}catch(_){}
  setStatus('Hand 3D em modo imersivo. Use a pinça para controlar a peça.','ok');
}

function v118ExitImmersive(){
  if(!v118Immersive)return;
  v118Immersive=false;
  v118ControlsVisible=true;
  document.body.classList.remove('v118-hand-immersive','v118-hand-controls-hidden');
  try{renderer?.setSize?.(stage.clientWidth,stage.clientHeight,false);}catch(_){}
}
