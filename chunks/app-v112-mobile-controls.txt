const v112IsTouch = matchMedia('(pointer: coarse)').matches || navigator.maxTouchPoints>0;
let v112HoldTimer=null;

function v112MobileParts(){
  if(typeof v111SelectedParts==='function') return v111SelectedParts().filter(p=>p.visible&&!p.locked);
  const ids=(typeof multiSelectedIds!=='undefined'&&multiSelectedIds.size)?[...multiSelectedIds]:(selectedPartId?[selectedPartId]:[]);
  return ids.map(id=>parts.find(p=>p.id===id)).filter(p=>p&&p.visible&&!p.locked);
}

function v112MoveStep(items){
  if(typeof v111SnapEnabled!=='undefined' && v111SnapEnabled) return Math.max(0.001,v111SnapStep||0.1);
  if(typeof selectionMoveStep==='function') return selectionMoveStep(items,false);
  const box=new THREE.Box3(); items.forEach(p=>box.expandByObject(p.mesh));
  const size=box.isEmpty()?new THREE.Vector3(1,1,1):box.getSize(new THREE.Vector3());
  return Math.min(Math.max(Math.max(size.x,size.y,size.z,1)*0.02,0.02),1);
}

function v112RecordMoveUndo(){
  if(typeof captureTransformUndo==='function' && typeof editorPushUndo==='function'){
    const a=captureTransformUndo('movimento mobile'); if(a) editorPushUndo(a);
  }
}

function v112Nudge(dir,record=true){
  const items=v112MobileParts(); if(!items.length){ if(record)setStatus('Selecione uma peça para mover.','warn'); return; }
  if(record) v112RecordMoveUndo();
  const step=v112MoveStep(items);
  const right=new THREE.Vector3(1,0,0).applyQuaternion(camera.quaternion).normalize();
  const up=new THREE.Vector3(0,1,0).applyQuaternion(camera.quaternion).normalize();
  const forward=new THREE.Vector3(); camera.getWorldDirection(forward).normalize();
  const delta=new THREE.Vector3();
  if(dir==='left') delta.addScaledVector(right,-step);
  if(dir==='right') delta.addScaledVector(right,step);
  if(dir==='up') delta.addScaledVector(up,step);
  if(dir==='down') delta.addScaledVector(up,-step);
  if(dir==='front') delta.addScaledVector(forward,step);
  if(dir==='back') delta.addScaledVector(forward,-step);
  items.forEach(p=>{
    p.mesh.position.add(delta);
    if(typeof v111SnapEnabled!=='undefined' && v111SnapEnabled){
      const s=Math.max(0.001,v111SnapStep||0.1);
      p.mesh.position.x=Math.round(p.mesh.position.x/s)*s;
      p.mesh.position.y=Math.round(p.mesh.position.y/s)*s;
      p.mesh.position.z=Math.round(p.mesh.position.z/s)*s;
    }
    p.mesh.updateMatrixWorld(true);
  });
  syncTransformFields();
  if(typeof refreshSelectionHelpers==='function') refreshSelectionHelpers();
  if(typeof v111RefreshGizmo==='function') v111RefreshGizmo();
}

function v112BindHold(btn,dir){
  const stop=()=>{ if(v112HoldTimer){clearInterval(v112HoldTimer);v112HoldTimer=null;} };
  btn.addEventListener('pointerdown',e=>{
    e.preventDefault(); e.stopPropagation(); stop();
    v112Nudge(dir,true);
    v112HoldTimer=setInterval(()=>v112Nudge(dir,false),120);
    try{btn.setPointerCapture(e.pointerId)}catch(_){}
  });
  ['pointerup','pointercancel','lostpointercapture'].forEach(type=>btn.addEventListener(type,e=>{e.preventDefault();stop();}));
}

}
