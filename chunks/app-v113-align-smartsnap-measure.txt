// V10.13 — ALINHAR/DISTRIBUIR + SNAP INTELIGENTE + MEDIDAS REAIS
let v113SmartSnap=false;
let v113SmartTolerance=0.08;
let v113UnitName='cm';
let v113UnitScale=1;
let v113GizmoSnapBound=false;

function v113Selection(){
  if(typeof v111SelectedParts==='function') return v111SelectedParts().filter(p=>p.visible&&!p.locked);
  const ids=(typeof multiSelectedIds!=='undefined'&&multiSelectedIds.size)?[...multiSelectedIds]:(selectedPartId?[selectedPartId]:[]);
  return ids.map(id=>parts.find(p=>p.id===id)).filter(p=>p&&p.visible&&!p.locked);
}

function v113BoxForPart(p){
  p.mesh.updateMatrixWorld(true);
  return new THREE.Box3().setFromObject(p.mesh);
}

function v113SelectionBox(sel=v113Selection()){
  const box=new THREE.Box3(); sel.forEach(p=>box.union(v113BoxForPart(p))); return box;
}

function v113PushTransformUndo(label){
  if(typeof captureTransformUndo==='function'&&typeof editorPushUndo==='function'){
    const a=captureTransformUndo(label); if(a) editorPushUndo(a);
  }
}

function v113AfterTransform(msg){
  syncTransformFields();
  if(typeof refreshSelectionHelpers==='function') refreshSelectionHelpers();
  if(typeof v111RefreshGizmo==='function') v111RefreshGizmo();
  v113UpdateMeasure();
  if(msg) setStatus(msg,'ok');
}

function v113Align(axis,mode){
  const sel=v113Selection(); if(sel.length<2){setStatus('Selecione pelo menos 2 objetos para alinhar.','warn');return;}
  v113PushTransformUndo('alinhar objetos');
  const boxes=sel.map(p=>({p,b:v113BoxForPart(p)}));
  const overall=v113SelectionBox(sel); const center=overall.getCenter(new THREE.Vector3());
  let target=mode==='min'?overall.min[axis]:mode==='max'?overall.max[axis]:center[axis];
  boxes.forEach(({p,b})=>{
    const c=b.getCenter(new THREE.Vector3())[axis];
    const value=mode==='min'?b.min[axis]:mode==='max'?b.max[axis]:c;
    p.mesh.position[axis]+=target-value;
  });
  v113AfterTransform(`${sel.length} objetos alinhados.`);
}

function v113Distribute(axis){
  const sel=v113Selection(); if(sel.length<3){setStatus('Selecione pelo menos 3 objetos para distribuir.','warn');return;}
  const entries=sel.map(p=>({p,c:v113BoxForPart(p).getCenter(new THREE.Vector3())[axis]})).sort((a,b)=>a.c-b.c);
  const first=entries[0].c,last=entries[entries.length-1].c,step=(last-first)/(entries.length-1);
  v113PushTransformUndo('distribuir objetos');
  entries.forEach((e,i)=>{e.p.mesh.position[axis]+=first+step*i-e.c;});
  v113AfterTransform(`${sel.length} objetos distribuídos igualmente.`);
}

function v113BestAxisSnap(selBox,targetBoxes,axis){
  const sc=selBox.getCenter(new THREE.Vector3())[axis];
  const svals=[selBox.min[axis],sc,selBox.max[axis]];
  let best=null;
  targetBoxes.forEach(tb=>{
    const tc=tb.getCenter(new THREE.Vector3())[axis];
    const tvals=[tb.min[axis],tc,tb.max[axis]];
    svals.forEach(s=>tvals.forEach(t=>{
      const d=t-s,ad=Math.abs(d);
      if(ad<=v113SmartTolerance && (!best||ad<best.ad)) best={d,ad};
    }));
  });
  return best?.d||0;
}

function v113ApplySmartSnap(source='movimento'){
  if(!v113SmartSnap) return false;
  const sel=v113Selection(); if(!sel.length) return false;
  const ids=new Set(sel.map(p=>p.id));
  const targets=parts.filter(p=>p.visible&&!ids.has(p.id)).map(v113BoxForPart);
  if(!targets.length) return false;
  const box=v113SelectionBox(sel);
  const delta=new THREE.Vector3(
    v113BestAxisSnap(box,targets,'x'),
    v113BestAxisSnap(box,targets,'y'),
    v113BestAxisSnap(box,targets,'z')
  );
  if(delta.lengthSq()<1e-12) return false;
  v113PushTransformUndo('snap inteligente');
  sel.forEach(p=>p.mesh.position.add(delta));
  v113AfterTransform(`Snap inteligente: ${source} encaixado em outro objeto.`);
  return true;
}

function v113ToggleSmartSnap(){
  v113SmartSnap=!v113SmartSnap;
  const b=document.getElementById('v113SmartSnapBtn'); if(b){b.classList.toggle('active',v113SmartSnap);b.textContent=v113SmartSnap?'Snap inteligente ligado':'Snap inteligente desligado';}
  const mb=document.getElementById('v113MobileSmartSnap'); if(mb) mb.classList.toggle('active',v113SmartSnap);
  setStatus(`Snap inteligente ${v113SmartSnap?'ativado':'desativado'}.`,'ok');
}

function v113UpdateMeasure(){
  const out=document.getElementById('v113MeasureOut'); if(!out) return;
  const sel=v113Selection(); if(!sel.length){out.textContent='Nenhuma seleção';return;}
  const box=v113SelectionBox(sel),s=box.getSize(new THREE.Vector3());
  const f=v113UnitScale;
  out.textContent=`L ${ (s.x*f).toFixed(2) } ${v113UnitName} × A ${ (s.y*f).toFixed(2) } ${v113UnitName} × P ${ (s.z*f).toFixed(2) } ${v113UnitName}`;
}

function v113ReadUnits(){
  v113UnitScale=Math.max(0.000001,parseFloat(document.getElementById('v113UnitScale')?.value)||1);
  v113UnitName=document.getElementById('v113UnitName')?.value||'cm';
  v113UpdateMeasure();
}

  if(v113GizmoSnapBound||typeof v111TransformControls==='undefined'||!v111TransformControls) return;
  v113GizmoSnapBound=true; clearInterval(v113BindGizmo);
  v111TransformControls.addEventListener('dragging-changed',ev=>{if(!ev.value&&v113SmartSnap)setTimeout(()=>v113ApplySmartSnap('gizmo'),0);});
},250);
setTimeout(()=>clearInterval(v113BindGizmo),10000);
