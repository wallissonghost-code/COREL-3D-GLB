function serializeProject(){ return {version:'10.0', parts: parts.map(part=>{ const g=part.mesh.geometry; return {id:part.id, name:part.name, type:part.type, visible:part.visible, locked:part.locked, position:part.mesh.position.toArray(), rotation:[part.mesh.rotation.x, part.mesh.rotation.y, part.mesh.rotation.z], scale:part.mesh.scale.toArray(), material:{color:'#'+part.mesh.material.color.getHexString(), roughness:part.mesh.material.roughness, metalness:part.mesh.material.metalness}, geometry:{positions:Array.from(g.attributes.position.array), colors:Array.from(g.getAttribute('color').array), mask:Array.from(g.userData.mask||[])} }; }), selectedPartId }; }
function saveProject(){ const data = JSON.stringify(serializeProject(), null, 2); const blob = new Blob([data], {type:'application/json'}); const a=document.createElement('a'); a.href=URL.createObjectURL(blob); a.download='corel-3d-glb-v10-project.json'; a.click(); setTimeout(()=>URL.revokeObjectURL(a.href),1000); setStatus('Projeto salvo para continuar editando depois.', 'ok'); }
function clearAllParts(){ while(parts.length){ const p=parts.pop(); scene.remove(p.mesh); p.mesh.geometry.dispose(); p.mesh.material.dispose(); } selectedPartId=null; renderLayerList(); refreshUi(); syncTransformFields(); $('empty').style.display='grid'; }
async function loadProjectFile(file){ try{ const txt=await file.text(); const data=JSON.parse(txt); clearAllParts(); let maxId=0; for(const item of (data.parts||[])){ const g=new THREE.BufferGeometry(); g.setAttribute('position', new THREE.Float32BufferAttribute(item.geometry.positions,3)); g.setAttribute('color', new THREE.Float32BufferAttribute(item.geometry.colors,3)); g.userData.mask = new Float32Array(item.geometry.mask || new Array(item.geometry.positions.length/3).fill(0)); g.computeVertexNormals(); const mesh=new THREE.Mesh(g, new THREE.MeshStandardMaterial({color:item.material.color, roughness:item.material.roughness, metalness:item.material.metalness, vertexColors:true})); mesh.castShadow=true; mesh.receiveShadow=true; mesh.position.fromArray(item.position||[0,0,0]); mesh.rotation.set(...(item.rotation||[0,0,0])); mesh.scale.fromArray(item.scale||[1,1,1]); mesh.userData.baseGeometry = g.clone(); mesh.userData.baseMask = g.userData.mask.slice(); const part={id:item.id||partIdSeed++, name:item.name||'Camada', type:item.type||'mesh', mesh, visible:item.visible!==false, locked:!!item.locked}; mesh.visible=part.visible; parts.push(part); scene.add(mesh); maxId=Math.max(maxId, part.id); } partIdSeed=Math.max(partIdSeed, maxId+1); selectedPartId = data.selectedPartId && parts.some(p=>p.id===data.selectedPartId) ? data.selectedPartId : parts[0]?.id || null; renderLayerList(); refreshUi(); syncTransformFields(); $('empty').style.display = parts.length ? 'none' : 'grid'; setStatus('Projeto carregado com sucesso.', 'ok'); } catch(err){ console.error(err); setStatus('Falha ao abrir o projeto salvo.', 'error'); } }
