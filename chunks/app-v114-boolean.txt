// V10.14 — BOOLEAN REAL (UNIÃO / SUBTRAÇÃO / INTERSEÇÃO)
let v114CsgModule=null;
let v114Busy=false;

async function v114LoadCsg(){
  if(v114CsgModule) return v114CsgModule;
  setStatus('Carregando motor Boolean 3D...');
  v114CsgModule=await import('https://esm.sh/three-bvh-csg@0.0.17?external=three');
  return v114CsgModule;
}

function v114SelectedPair(){
  const ids=(typeof multiSelectedIds!=='undefined'&&multiSelectedIds.size)?[...multiSelectedIds]:(selectedPartId?[selectedPartId]:[]);
  const uniq=[...new Set(ids)].filter(id=>parts.some(p=>p.id===id));
  if(uniq.length!==2) return null;
  const aid=uniq.includes(selectedPartId)?selectedPartId:uniq[uniq.length-1];
  const bid=uniq.find(id=>id!==aid);
  const a=parts.find(p=>p.id===aid), b=parts.find(p=>p.id===bid);
  if(!a||!b) return null;
  return {a,b};
}

function v114CloneMaterial(mat){
  if(Array.isArray(mat)) return mat[0]?.clone?.() || makeMaterial();
  return mat?.clone?.() || makeMaterial();
}

function v114WorldGeometry(part){
  part.mesh.updateMatrixWorld(true);
  let g=part.mesh.geometry.clone();
  if(g.index && g.toNonIndexed) g=g.toNonIndexed();
  g.applyMatrix4(part.mesh.matrixWorld);
  g.computeVertexNormals();
  g.computeBoundingBox();
  g.computeBoundingSphere();
  return g;
}

function v114RecordUndo(label){
  if(typeof captureFullSceneSnapshot!=='function' || typeof originalEditorPushUndoV109!=='function') return;
  const action={label,beforeSnapshot:captureFullSceneSnapshot(),undo(){}};
  if(typeof editorRedoStack!=='undefined') editorRedoStack.length=0;
  originalEditorPushUndoV109(action);
  if(typeof logHistoryV109==='function') logHistoryV109(label,'do');
}

function v114SetOriginalVisibility(pair,visible){
  [pair.a,pair.b].forEach(p=>{p.visible=visible;p.mesh.visible=visible;});
}

async function v114Boolean(opName){
  if(v114Busy) return;
  const pair=v114SelectedPair();
  if(!pair){ setStatus('Boolean precisa de exatamente 2 objetos selecionados. O objeto ativo é a peça A.','warn'); return; }
  if(pair.a.locked||pair.b.locked){ setStatus('Desbloqueie as duas peças antes do Boolean.','warn'); return; }
  if(!pair.a.visible||!pair.b.visible){ setStatus('As duas peças precisam estar visíveis para o Boolean.','warn'); return; }

  v114Busy=true;
  const buttons=['v114Union','v114Subtract','v114Intersect']; buttons.forEach(id=>{const b=document.getElementById(id);if(b)b.disabled=true;});
  try{
    setStatus(`Calculando Boolean: ${opName}...`);
    const CSG=await v114LoadCsg();
    const {Brush,Evaluator,ADDITION,SUBTRACTION,INTERSECTION}=CSG;
    if(!Brush||!Evaluator) throw new Error('Motor CSG incompatível.');

    const ga=v114WorldGeometry(pair.a), gb=v114WorldGeometry(pair.b);
    if(!ga.attributes?.position?.count||!gb.attributes?.position?.count) throw new Error('Uma das peças não possui geometria válida.');

    const brushA=new Brush(ga,v114CloneMaterial(pair.a.mesh.material));
    const brushB=new Brush(gb,v114CloneMaterial(pair.b.mesh.material));
    brushA.updateMatrixWorld(true); brushB.updateMatrixWorld(true);
    const evaluator=new Evaluator();
    evaluator.useGroups=false;
    evaluator.consolidateGroups=true;
    const operation=opName==='uniao'?ADDITION:opName==='subtracao'?SUBTRACTION:INTERSECTION;
    const result=evaluator.evaluate(brushA,brushB,operation);
    const rg=result?.geometry?.clone?.();
    if(!rg?.attributes?.position?.count) throw new Error('A operação produziu uma malha vazia. Verifique se as peças se cruzam.');

    rg.computeVertexNormals(); rg.computeBoundingBox(); rg.computeBoundingSphere();
    const label=opName==='uniao'?'União':opName==='subtracao'?'Subtração':'Interseção';
    v114RecordUndo(`Boolean: ${label.toLowerCase()}`);

    const cam={pos:camera.position.clone(),quat:camera.quaternion.clone(),up:camera.up.clone(),target:controls.target.clone()};
    const out=createPartFromGeometry(rg,`${label}: ${pair.a.name} + ${pair.b.name}`,'boolean');
    try{out.mesh.material.dispose?.();}catch(_){}
    out.mesh.material=v114CloneMaterial(pair.a.mesh.material);
    out.mesh.position.set(0,0,0); out.mesh.quaternion.identity(); out.mesh.scale.set(1,1,1);
    out.mesh.userData.baseGeometry=out.mesh.geometry.clone();
    out.mesh.userData.baseMask=new Float32Array(out.mesh.geometry.attributes.position.count);

    const keep=document.getElementById('v114KeepOriginals')?.checked;
    if(!keep) v114SetOriginalVisibility(pair,false);

    if(typeof multiSelectedIds!=='undefined'){multiSelectedIds.clear();multiSelectedIds.add(out.id);} selectedPartId=out.id;
    camera.position.copy(cam.pos);camera.quaternion.copy(cam.quat);camera.up.copy(cam.up);controls.target.copy(cam.target);controls.update();
    renderLayerList();refreshUi();syncTransformFields();
    if(typeof refreshSelectionHelpers==='function')refreshSelectionHelpers();
    if(typeof v111RefreshGizmo==='function')v111RefreshGizmo();
    if(typeof v113UpdateMeasureUi==='function')v113UpdateMeasureUi();
    setStatus(`${label} concluída. Resultado criado como nova camada${keep?' mantendo':' ocultando'} as originais.`,'ok');
  }catch(err){
    console.error('Boolean V10.14:',err);
    setStatus('Falha no Boolean: '+(err?.message||err),'error');
  }finally{
    v114Busy=false; buttons.forEach(id=>{const b=document.getElementById(id);if(b)b.disabled=false;});
  }
}
