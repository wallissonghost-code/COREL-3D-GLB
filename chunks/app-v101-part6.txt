function cutterBasis(normal){
  const n = normal.clone().normalize();
  const ref = Math.abs(n.z) < 0.85 ? new THREE.Vector3(0,0,1) : new THREE.Vector3(0,1,0);
  const u = new THREE.Vector3().crossVectors(ref,n).normalize();
  const v = new THREE.Vector3().crossVectors(n,u).normalize();
  return {n,u,v};
}

function hideCutPreview(){
  if(cutPreview){ if(cutPreview.parent) cutPreview.parent.remove(cutPreview); cutPreview.geometry.dispose(); cutPreview.material.dispose(); cutPreview=null; }
}

function showCutPreview(hit){
  const mesh = currentMesh(); if(!mesh || !hit) return;
  hideCutPreview();
  const center = mesh.worldToLocal(hit.point.clone());
  const {n,u,v} = cutterBasis(hit.face.normal);
  const w = +$('cutWidth').value, h = +$('cutHeight').value;
  const pts=[];
  if($('cutShape').value==='circle'){
    const r=w*0.5;
    for(let i=0;i<48;i++){ const a=i/48*Math.PI*2; pts.push(center.clone().addScaledVector(u,Math.cos(a)*r).addScaledVector(v,Math.sin(a)*r).addScaledVector(n,0.006)); }
  } else {
    const hw=w*0.5, hh=h*0.5;
    pts.push(
      center.clone().addScaledVector(u,-hw).addScaledVector(v,-hh).addScaledVector(n,0.006),
      center.clone().addScaledVector(u, hw).addScaledVector(v,-hh).addScaledVector(n,0.006),
      center.clone().addScaledVector(u, hw).addScaledVector(v, hh).addScaledVector(n,0.006),
      center.clone().addScaledVector(u,-hw).addScaledVector(v, hh).addScaledVector(n,0.006)
    );
  }
  const g=new THREE.BufferGeometry().setFromPoints(pts);
  cutPreview=new THREE.LineLoop(g,new THREE.LineBasicMaterial({color:0xffa640,depthTest:false,transparent:true,opacity:0.95}));
  cutPreview.renderOrder=999;
  mesh.add(cutPreview);
}

function flatCutStamp(center, stroke){
  const mesh=currentMesh(); if(!mesh || !stroke) return;
  const geom=mesh.geometry, pos=geom.attributes.position, arr=pos.array;
  const mask=geom.userData.mask || new Float32Array(pos.count);
  const w=+$('cutWidth').value, h=+$('cutHeight').value, depth=+$('cutDepth').value;
  const shape=$('cutShape').value, hw=w*0.5, hh=h*0.5;
  const eps=Math.max(0.025,depth*0.12);
  let changed=0;
  for(let i=0;i<pos.count;i++){
    if(mask[i]>0.5) continue;
    const vtx=new THREE.Vector3(arr[i*3],arr[i*3+1],arr[i*3+2]);
    const d=vtx.clone().sub(center);
    const du=d.dot(stroke.u), dv=d.dot(stroke.v);
    const inside = shape==='circle' ? (du*du+dv*dv <= hw*hw) : (Math.abs(du)<=hw && Math.abs(dv)<=hh);
    if(!inside) continue;
    const planeCoord=vtx.dot(stroke.n);
    if(planeCoord < stroke.targetC-eps || planeCoord > stroke.baseC+eps) continue;
    const shift=stroke.targetC-planeCoord;
    vtx.addScaledVector(stroke.n,shift);
    arr[i*3]=vtx.x; arr[i*3+1]=vtx.y; arr[i*3+2]=vtx.z;
    changed++;
  }
  if(changed){
    pos.needsUpdate=true;
    geom.computeVertexNormals();
    geom.computeBoundingBox();
    geom.computeBoundingSphere();
  }
  return changed;
}

function beginFlatCut(hit,e){
  const mesh=currentMesh(); if(!mesh || !hit) return false;
  pushUndo();
  const origin=mesh.worldToLocal(hit.point.clone());
  const basis=cutterBasis(hit.face.normal);
  const depth=+$('cutDepth').value;
  const baseC=origin.dot(basis.n);
  cutStroke={...basis,baseC,targetC:baseC-depth,lastCenter:origin.clone()};
  cutDragging=true;
  const changed=flatCutStamp(origin,cutStroke);
  try{ renderer.domElement.setPointerCapture(e.pointerId); }catch(_){}
  setStatus(changed ? 'Cortando • fundo plano travado na profundidade escolhida.' : 'Nenhum vértice dentro do corte. Use Mais detalhe para aumentar a malha.', changed?'ok':'warn');
  return true;
}

function continueFlatCut(hit){
  if(!cutDragging || !cutStroke || !hit) return;
  const mesh=currentMesh();
  let center=mesh.worldToLocal(hit.point.clone());
  const dc=center.dot(cutStroke.n)-cutStroke.baseC;
  center.addScaledVector(cutStroke.n,-dc);
  const spacing=Math.max(0.04,Math.min(+$('cutWidth').value,+$('cutHeight').value)*0.22);
  if(center.distanceTo(cutStroke.lastCenter)<spacing) return;
  const dist=center.distanceTo(cutStroke.lastCenter);
  const steps=Math.max(1,Math.ceil(dist/spacing));
  const start=cutStroke.lastCenter.clone();
  for(let s=1;s<=steps;s++) flatCutStamp(start.clone().lerp(center,s/steps),cutStroke);
  cutStroke.lastCenter.copy(center);
}

async function importGlbFile(file){
  setStatus('Importando GLB e preparando malhas para edição...');
  try{
    const { GLTFLoader } = await import('three/addons/loaders/GLTFLoader.js');
    const buffer=await file.arrayBuffer();
    const loader=new GLTFLoader();
    const gltf=await new Promise((resolve,reject)=>loader.parse(buffer,'',resolve,reject));
    gltf.scene.updateMatrixWorld(true);
    const imported=[];
    gltf.scene.traverse(node=>{
      if(!node.isMesh || !node.geometry?.attributes?.position) return;
      const geometry=node.geometry.clone();
      geometry.applyMatrix4(node.matrixWorld);
      const base=file.name.replace(/\.glb$/i,'');
      const part=createPartFromGeometry(geometry, node.name ? `${base} • ${node.name}` : `${base} • peça ${imported.length+1}`, 'glb');
      const sourceMat=Array.isArray(node.material)?node.material[0]:node.material;
      if(sourceMat?.isMaterial){
        try{
          const mat=sourceMat.clone();
          if('vertexColors' in mat) mat.vertexColors=true;
          part.mesh.material.dispose();
          part.mesh.material=mat;
        }catch(_){}
      }
      imported.push(part);
    });
    if(!imported.length) throw new Error('O GLB não contém malhas editáveis.');
    selectPart(imported[0].id);
    $('empty').style.display='none';
    renderLayerList(); refreshUi(); syncTransformFields(); fitCameraToSelection();
    setStatus(`GLB importado: ${imported.length} camada${imported.length===1?'':'s'} editável${imported.length===1?'':'is'}.`, 'ok');
  }catch(err){
    console.error('Erro ao importar GLB:',err);
    setStatus('Falha ao importar GLB: '+(err?.message||err), 'error');
  }
}
