async function importSvgFile(file){
  try{
    const txt = await file.text(); const loader = new SVGLoader(); const data = loader.parse(txt); const geos = [];
    data.paths.forEach(path=>{ const shapes = SVGLoader.createShapes(path); shapes.forEach(shape=>{ const g = new THREE.ExtrudeGeometry(shape,{depth:0.15,bevelEnabled:false,curveSegments:10}); geos.push(g.toNonIndexed()); }); });
    if(!geos.length){ setStatus('Nenhuma forma preenchida foi encontrada no SVG.', 'error'); return; }
    const merged = mergeGeometries(geos, false); merged.center(); merged.rotateX(Math.PI);
    createPartFromGeometry(merged, file.name.replace(/\.svg$/i,''), 'svg');
  }catch(err){ console.error(err); setStatus('Falha ao importar SVG.', 'error'); }
}

async function importImageRelief(file){
  try{
    const img = await createImageBitmap(file);
    const maxSide = 80;
    let w = img.width, h = img.height;
    const scale = Math.min(1, maxSide / Math.max(w,h));
    w = Math.max(12, Math.round(w*scale));
    h = Math.max(12, Math.round(h*scale));
    const canvas = document.createElement('canvas'); canvas.width = w; canvas.height = h;
    const ctx = canvas.getContext('2d', {willReadFrequently:true});
    ctx.drawImage(img, 0, 0, w, h);
    const data = ctx.getImageData(0,0,w,h).data;
    const geom = new THREE.PlaneGeometry(2, 2*(h/w), w-1, h-1);
    const pos = geom.attributes.position;
    for(let iy=0; iy<h; iy++){
      for(let ix=0; ix<w; ix++){
        const src=((h-1-iy)*w+ix)*4;
        const r=data[src], g=data[src+1], b=data[src+2], a=data[src+3]/255;
        const lum=((r+g+b)/3)/255;
        const depth=(lum*0.42 + a*0.08);
        const idx=iy*w+ix;
        pos.setZ(idx, depth);
      }
    }
    geom.computeVertexNormals();
    geom.center();
    createPartFromGeometry(geom, file.name.replace(/\.(png|jpg|jpeg)$/i,''), 'image-relief');
    setStatus('PNG/JPG importado como relevo 3D simplificado.', 'ok');
  }catch(err){ console.error(err); setStatus('Falha ao importar PNG/JPG.', 'error'); }
}
