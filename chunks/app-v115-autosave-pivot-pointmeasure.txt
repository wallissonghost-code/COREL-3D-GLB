function v115OpenDb(){
  return new Promise((resolve,reject)=>{
    const req=indexedDB.open(V115_DB,1);
    req.onupgradeneeded=()=>{ if(!req.result.objectStoreNames.contains(V115_STORE)) req.result.createObjectStore(V115_STORE); };
    req.onsuccess=()=>resolve(req.result); req.onerror=()=>reject(req.error);
  });
}
function v115Tx(mode,fn){
  return v115OpenDb().then(db=>new Promise((resolve,reject)=>{
    const tx=db.transaction(V115_STORE,mode), store=tx.objectStore(V115_STORE);
    let req; try{req=fn(store);}catch(e){db.close();reject(e);return;}
    tx.oncomplete=()=>{db.close();resolve(req?.result);}; tx.onerror=()=>{db.close();reject(tx.error);};
  }));
}
function v115MaterialData(mat){
  const m=Array.isArray(mat)?mat[0]:mat;
  return {
    type:m?.type||'MeshStandardMaterial', color:m?.color?.getHex?.()??0x7c5cff,
    roughness:Number.isFinite(m?.roughness)?m.roughness:.7, metalness:Number.isFinite(m?.metalness)?m.metalness:.04,
    opacity:Number.isFinite(m?.opacity)?m.opacity:1, transparent:!!m?.transparent,
    side:m?.side??THREE.FrontSide, vertexColors:m?.vertexColors!==false,
    emissive:m?.emissive?.getHex?.()??0, emissiveIntensity:Number.isFinite(m?.emissiveIntensity)?m.emissiveIntensity:1
  };
}
function v115MakeMaterial(d){
  return new THREE.MeshStandardMaterial({color:d?.color??0x7c5cff,roughness:d?.roughness??.7,metalness:d?.metalness??.04,opacity:d?.opacity??1,transparent:!!d?.transparent,side:d?.side??THREE.FrontSide,vertexColors:d?.vertexColors!==false,emissive:d?.emissive??0,emissiveIntensity:d?.emissiveIntensity??1});
}
function v115GeometryData(g){
  const attrs={};
  for(const [name,a] of Object.entries(g.attributes||{})) attrs[name]={array:a.array.slice? a.array.slice():new a.array.constructor(a.array),itemSize:a.itemSize,normalized:!!a.normalized};
  return {attrs,index:g.index?{array:g.index.array.slice?g.index.array.slice():new g.index.array.constructor(g.index.array)}:null,groups:(g.groups||[]).map(x=>({...x})),drawRange:{...g.drawRange},mask:g.userData?.mask?new Float32Array(g.userData.mask):null};
}
function v115GeometryFromData(d){
  const g=new THREE.BufferGeometry();
  for(const [name,a] of Object.entries(d?.attrs||{})) g.setAttribute(name,new THREE.BufferAttribute(a.array,a.itemSize,a.normalized));
  if(d?.index?.array) g.setIndex(new THREE.BufferAttribute(d.index.array,1));
  (d?.groups||[]).forEach(x=>g.addGroup(x.start,x.count,x.materialIndex));
  if(d?.drawRange) g.setDrawRange(d.drawRange.start,d.drawRange.count);
  if(d?.mask) g.userData.mask=new Float32Array(d.mask);
  if(!g.getAttribute('normal')&&g.getAttribute('position')) g.computeVertexNormals();
  initVertexAttributes(g); return g;
}
