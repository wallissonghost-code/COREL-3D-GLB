function applyMaskAtPoint(localPoint, radius, erase=false){ const mesh=currentMesh(), geom=mesh.geometry, pos=geom.attributes.position.array, mask=geom.userData.mask, colors=geom.getAttribute('color').array, falloff=+$('brushFalloff').value; for(let i=0;i<mask.length;i++){ const v=new THREE.Vector3(pos[i*3],pos[i*3+1],pos[i*3+2]); const d=v.distanceTo(localPoint); if(d>radius) continue; const t=1-d/radius, w=Math.pow(Math.max(0,t),falloff), delta=w*0.35; mask[i]=THREE.MathUtils.clamp(mask[i]+(erase?-delta:delta),0,1); const c=grayColor(mask[i]); colors[i*3]=c.r; colors[i*3+1]=c.g; colors[i*3+2]=c.b; } geom.getAttribute('color').needsUpdate=true; updateMaskStats(); }
function sculptAt(localPoint, localNormal, forceScale=1){ const mesh=currentMesh(), geom=mesh.geometry, pos=geom.attributes.position, arr=pos.array, mask=geom.userData.mask; const radius=+$('brushSize').value*0.018, strength=+$('brushStrength').value*forceScale, falloff=+$('brushFalloff').value, mode=evtShiftSmooth?'smooth':$('brushMode').value; const positions=[]; for(let i=0;i<arr.length;i+=3) positions.push(new THREE.Vector3(arr[i],arr[i+1],arr[i+2])); if(maskMode==='paint' || maskMode==='erase'){ applyMaskAtPoint(localPoint, radius, maskMode==='erase'); return; } for(let i=0;i<positions.length;i++){ if(mask[i]>0.5) continue; const v=positions[i], d=v.distanceTo(localPoint); if(d>radius) continue; const t=1-d/radius, w=Math.pow(Math.max(0,t), falloff); if(mode==='clay'){ v.add(localNormal.clone().multiplyScalar(strength*w*0.20)); const toward=localPoint.clone().sub(v); toward.sub(localNormal.clone().multiplyScalar(toward.dot(localNormal))); v.add(toward.multiplyScalar(strength*w*0.018)); continue; } if(mode==='smooth'){ let avg=new THREE.Vector3(), count=0; for(let j=0;j<positions.length;j++){ if(i===j || mask[j]>0.5) continue; const vd=positions[j].distanceTo(v); if(vd<radius*0.55){ avg.add(positions[j]); count++; } } if(count>0) v.lerp(avg.multiplyScalar(1/count), strength*w*0.35); continue; } if(mode==='flatten'){ const planeDelta=localPoint.clone().sub(v).dot(localNormal); v.add(localNormal.clone().multiplyScalar(planeDelta*strength*w*0.72)); continue; } if(mode==='pinch'){ const dir=localPoint.clone().sub(v); v.add(dir.multiplyScalar(strength*w*0.34)); continue; } if(mode==='crease'){ const dir=localPoint.clone().sub(v).normalize(); v.add(dir.multiplyScalar(strength*w*0.16)); v.add(localNormal.clone().multiplyScalar(-strength*w*0.12)); continue; } let sign=1; if(mode==='lower') sign=-1; if(mode==='inflate') sign=1.6; const useNormal=mode==='inflate'?v.clone().normalize():localNormal; v.add(useNormal.clone().multiplyScalar(sign*strength*w*0.18)); } for(let i=0;i<positions.length;i++){ arr[i*3]=positions[i].x; arr[i*3+1]=positions[i].y; arr[i*3+2]=positions[i].z; } pos.needsUpdate=true; geom.computeVertexNormals(); }
function doStrokeAtHit(hit){ const mesh=currentMesh(); const localPoint=mesh.worldToLocal(hit.point.clone()); const localNormal=hit.face.normal.clone().normalize(); sculptAt(localPoint, localNormal); const mirrors=[]; if(symmetry.x) mirrors.push(new THREE.Vector3(-localPoint.x, localPoint.y, localPoint.z)); if(symmetry.y) mirrors.push(new THREE.Vector3(localPoint.x, -localPoint.y, localPoint.z)); if(symmetry.z) mirrors.push(new THREE.Vector3(localPoint.x, localPoint.y, -localPoint.z)); if(symmetry.x&&symmetry.y) mirrors.push(new THREE.Vector3(-localPoint.x,-localPoint.y,localPoint.z)); if(symmetry.x&&symmetry.z) mirrors.push(new THREE.Vector3(-localPoint.x,localPoint.y,-localPoint.z)); if(symmetry.y&&symmetry.z) mirrors.push(new THREE.Vector3(localPoint.x,-localPoint.y,-localPoint.z)); if(symmetry.x&&symmetry.y&&symmetry.z) mirrors.push(new THREE.Vector3(-localPoint.x,-localPoint.y,-localPoint.z)); const done=new Set(); mirrors.forEach(p=>{ const key=`${p.x.toFixed(4)}|${p.y.toFixed(4)}|${p.z.toFixed(4)}`; if(done.has(key)) return; done.add(key); sculptAt(p, localNormal); }); }
function maybeDoStroke(hit){ const current=hit.point.clone(), radius=+$('brushSize').value*0.018, spacing=+$('brushSpacing').value; if(!lastStrokePoint){ doStrokeAtHit(hit); lastStrokePoint=current; strokeDirty=true; return; } const minStep=radius*spacing, dist=current.distanceTo(lastStrokePoint); if(dist<minStep) return; const steps=Math.max(1, Math.floor(dist/minStep)); for(let i=1;i<=steps;i++){ const p=lastStrokePoint.clone().lerp(current,i/steps); doStrokeAtHit({point:p, face:hit.face}); } lastStrokePoint=current; strokeDirty=true; }
